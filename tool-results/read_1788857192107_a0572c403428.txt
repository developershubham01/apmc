    71→    71→function relativeTime(iso: string): string {
    72→    72→  const then = new Date(iso).getTime();
    73→    73→  const diff = Date.now() - then;
    74→    74→  const mins = Math.floor(diff / 60000);
    75→    75→  if (mins < 1) return "just now";
    76→    76→  if (mins < 60) return `${mins} min ago`;
    77→    77→  const hours = Math.floor(mins / 60);
    78→    78→  if (hours < 24) return `${hours} hr ago`;
    79→    79→  const days = Math.floor(hours / 24);
    80→    80→  if (days < 7) return `${days} day${days > 1 ? "s" : ""} ago`;
    81→    81→  return new Date(iso).toLocaleDateString("en-IN", {
    82→    82→    day: "numeric",
    83→    83→    month: "short",
    84→    84→    year: "numeric",
    85→    85→  });
    86→    86→}
